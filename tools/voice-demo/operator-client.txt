const screen=document.querySelector('#screen'),status=document.querySelector('#status'),message=document.querySelector('#message');let busy=false,old;
async function action(name,data={}){try{const r=await fetch('/action/'+name,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(data)});message.textContent=r.ok?'Ausgeführt.':await r.text()}catch{message.textContent='Verbindung unterbrochen. Status prüfen.'}}
document.querySelector('#prepare').onclick=()=>action('prepare');document.querySelector('#connect').onclick=()=>action('connect');document.querySelector('#stop').onclick=()=>action('stop');document.querySelector('#tone').onclick=()=>action('tone');document.querySelector('#up').onclick=()=>action('scroll',{y:-600});document.querySelector('#down').onclick=()=>action('scroll',{y:600});screen.onclick=e=>{const b=screen.getBoundingClientRect();action('click',{x:(e.clientX-b.left)*1100/b.width,y:(e.clientY-b.top)*900/b.height})};
async function refresh(){if(busy)return;busy=true;try{const r=await fetch('/status');const s=await r.json();status.textContent='Interview: '+s.interview+'\nLUKAS: '+s.lukas.state+'\nSicherheitsprüfung: '+(s.checks.security_check_complete?'erledigt':'noch offen')+'\nBrowser verfügbar: '+s.browser_open+'\nZugang läuft in '+Math.ceil(s.expires_in_seconds/60)+' Minuten ab.';document.querySelector('#connect').disabled=s.armed||s.attempt_used||!s.browser_open||!s.prepared||!s.checks.security_check_complete;if(s.browser_open){const image=await fetch('/screen');if(image.ok){const url=URL.createObjectURL(await image.blob());screen.src=url;if(old)URL.revokeObjectURL(old);old=url}}}catch{message.textContent='Status nicht erreichbar. Die Seite wird weiter geprüft.'}finally{busy=false}}refresh();setInterval(refresh,1400);
